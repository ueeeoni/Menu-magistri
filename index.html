<script>
// Dati di default iniziali se il
database locale è vuoto
const defaultDishes = [
{
id: 1,
name: "Anchette di Pollo alla
Brace",
category: "Braceria",
price: "€ 8,00",
desc: "Pollo marinato con
erbe aromatiche e cotto alla brace.",
image: ""
},
{
id: 2,
name: "Tortiglioni al Ragù",
category: "Primi",
price: "€ 7,00",
desc: "Pasta fresca con ragù
tradizionale a lunga cottura.",
image: ""
}
];
// Carica i dati salvati oppure usa
quelli di default
let dishes =
JSON.parse(localStorage.getItem('ma-
gistri_menu')) || defaultDishes;
let currentCategory = 'Tutti';
function saveAndRender() {
localStorage.setItem('magistri_menu',
JSON.stringify(dishes));
renderMenu();
renderAdminList();
}
function renderMenu() {
const grid =
document.getElementById('menuGrid')
;
grid.innerHTML = '';
const filtered = currentCategory
=== 'Tutti'
? dishes
: dishes.filter(d => d.category
=== currentCategory);
if (filtered.length === 0) {
grid.innerHTML = '<p
style="grid-column: 1/-1; text-
align:center; color:#777;">Nessun
piatto presente in questa categoria.</
p>';
return;
}
filtered.forEach(dish => {
const card =
document.createElement('div');
card.className = 'menu-
card';
// Se non c'è immagine carica
un segnaposto grigio
const imgSrc = dish.image ||
'data:image/svg+xml;utf8,<svg
xmlns="http://www.w3.org/2000/svg"
width="100" height="100" viewBox="0
0 24 24" fill="none" stroke="%23ccc"
stroke-width="2"><rect x="3" y="3"
width="18" height="18" rx="2" ry="2"/
><circle cx="8.5" cy="8.5" r="1.5"/
><polyline points="21 15 16 10 5 21"/
></svg>';
card.innerHTML = `
<img src="${imgSrc}"
alt="${dish.name}">
<div class="card-
content">
<h3 class="card-title">$
{dish.name}</h3>
<p class="card-desc">$
{dish.desc || ''}</p>
<div class="card-
footer">
<span class="card-
price">${dish.price}</span>
</div>
</div>
`;
grid.appendChild(card);
});
}
function filterCategory(cat, btn) {
currentCategory = cat;
document.querySelectorAll('.category-
btn').forEach(b =>
b.classList.remove('active'));
btn.classList.add('active');
renderMenu();
}
function toggleAdmin() {
const panel =
document.getElementById('admin-
panel');
panel.style.display =
panel.style.display === 'block' ?
'none' : 'block';
}
function addDish() {
const name =
document.getElementById('newName')
.value.trim();
const category =
document.getElementById('newCate-
gory').value;
const price =
document.getElementById('newPrice').
value.trim();
const desc =
document.getElementById('newDesc').
value.trim();
const imageInput =
document.getElementById('newImage'
);
if (!name || !price) {
alert("Inserisci almeno il
nome e il prezzo del piatto.");
return;
}
const newDishObj = {
id: Date.now(),
name,
category,
price: price.includes('€') ?
price : '€ ' + price,
desc,
image: ''
};
// Gestione caricamento
immagine locale
if (imageInput.files &&
imageInput.files[0]) {
const reader = new
FileReader();
reader.onload = function(e) {
newDishObj.image =
e.target.result;
dishes.push(newDishObj);
saveAndRender();
clearForm();
};
reader.readAsDataURL(imageInput.files
[0]);
} else {
dishes.push(newDishObj);
saveAndRender();
clearForm();
}
}
function clearForm() {
document.getElementById('newName')
.value = '';
document.getElementById('newPrice').
value = '';
document.getElementById('newDesc').
value = '';
document.getElementById('newImage'
).value = '';
}
function deleteDish(id) {
if (confirm("Vuoi davvero
eliminare questo piatto?")) {
dishes = dishes.filter(d =>
d.id !== id);
saveAndRender();
}
}
function renderAdminList() {
const list =
document.getElementById('admin-
dishes-list');
list.innerHTML = '';
dishes.forEach(dish => {
const row =
document.createElement('div');
row.className = 'admin-
item-row';
row.innerHTML = `
<span><b>${dish.name}</
b> (${dish.category}) - ${dish.price}</
span>
<button class="btn btn-
danger" onclick="deleteDish($
{dish.id})">Elimina</button>
`;
list.appendChild(row);
});
}
// Avvio iniziale
renderMenu();
renderAdminList();
</script>
